public class problem23 {
    public static void main(String[] args) {

    }

    //采用分治的思想
    public ListNode mergeKLists(ListNode[] lists) {
        //如果要合并的内容为空，直接返回null
        if (lists == null || lists.length == 0) {
            return null;
        }

        //合并第一个到最后一个列表
        return mergeHelper(lists, 0, lists.length - 1);
    }

    //这里是分，把要处理的问题划分到最小的单位为止
    private ListNode mergeHelper(ListNode[] lists, int left, int right) {
        //如果分到了只剩一个，那么就直接返回，因为一个的话，它就已经是有序的了
        if (left == right) {
            return lists[left];
        }

        int mid = (left + right) / 2;
        //如果还没有分发到最小的单位，那么就继续从中点位置再分
        ListNode leftList = mergeHelper(lists, left, mid);
        ListNode rightList = mergeHelper(lists, mid + 1, right);

        //上面的得到的已经是最小的单位了，分完了之后就要治理了
        return mergeTwoLists(leftList, rightList);
    }

    private ListNode mergeTwoLists(ListNode l1, ListNode l2) {
        ListNode dummy = new ListNode(0);
        ListNode curr = dummy;
        while (l1 != null && l2 != null) {
            if (l1.val < l2.val) {
                curr.next = l1;
                l1 = l1.next;
            } else {
                curr.next = l2;
                l2 = l2.next;
            }
            curr = curr.next;
        }
        curr.next = l1 == null ? l2 : l1;
        return dummy.next;
    }


}
