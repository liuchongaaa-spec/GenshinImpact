class DeNode{
    int key;
    int value;
    DeNode next;
    DeNode pre;

    public DeNode(){}

    public DeNode(int key, int value){
        this.key = key;
        this.value = value;
    }
}

class LRUCache {

    Map<Integer, DeNode> map;
    DeNode head, tail;
    int capacity;

    public LRUCache(int capacity) {
        map = new HashMap<>();
        this.capacity = capacity;
        head = new DeNode();
        tail = new DeNode();
        head.next = tail;
        tail.pre = head;
    }

    public int get(int key) {
        if(map.containsKey(key)){
            DeNode node = map.get(key);
            remove(node);
            addToFirst(node);
            return node.value;
        }else{
            return -1;
        }
    }

    public void put(int key, int value) {
        if(map.containsKey(key)){
            DeNode node = map.get(key);
            node.value = value;
            remove(node);
            addToFirst(node);
        }else{
            DeNode node = new DeNode(key, value);
            if(map.size() == capacity){
                map.remove(tail.pre.key);
                remove(tail.pre);
            }
            addToFirst(node);
            map.put(key, node);
        }
    }

    private void remove(DeNode node){
        node.pre.next = node.next;
        node.next.pre = node.pre;
    }

    private void addToFirst(DeNode node){
        node.next = head.next;
        node.pre = head;
        node.next.pre = node;
        node.pre.next = node;
    }
}

   - 如果音频是在追问截图中的题目，例如补充限制条件、规定解题方法、询问其他方法、分析复杂度、指出错误或要求修改代码，则结合音频与截图给出最新答案。

   - 如果音频与截图的关系不够明确，优先依据面试官最近且完整的表达以及对话连续性判断当前题目，不要同时回答两道不同的算法题。
