class DeNode{
    int key;
    int value;
    DeNode next;
    DeNode pre;

    public DeNode(){}

    public DeNode(int key, int value){
        this.key = key;
        this.value = value;
    }
}

class LRUCache {

    Map<Integer, DeNode> map;
    DeNode head, tail;
    int capacity;

    public LRUCache(int capacity) {
        map = new HashMap<>();
        this.capacity = capacity;
        head = new DeNode();
        tail = new DeNode();
        head.next = tail;
        tail.pre = head;
    }

    public int get(int key) {
        if(map.containsKey(key)){
            DeNode node = map.get(key);
            remove(node);
            addToFirst(node);
            return node.value;
        }else{
            return -1;
        }
    }

    public void put(int key, int value) {
        if(map.containsKey(key)){
            DeNode node = map.get(key);
            node.value = value;
            remove(node);
            addToFirst(node);
        }else{
            DeNode node = new DeNode(key, value);
            if(map.size() == capacity){
                map.remove(tail.pre.key);
                remove(tail.pre);
            }
            addToFirst(node);
            map.put(key, node);
        }
    }

    private void remove(DeNode node){
        node.pre.next = node.next;
        node.next.pre = node.pre;
    }

    private void addToFirst(DeNode node){
        node.next = head.next;
        node.pre = head;
        node.next.pre = node;
        node.pre.next = node;
    }
}
